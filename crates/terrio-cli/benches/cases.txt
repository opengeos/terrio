# Reprojection, EPSG:4326 to UTM zone 31N.
warp_near warp -r near --t-srs EPSG:32631 --tr 360 360 {dem4326} {out}.tif
warp_bilinear warp -r bilinear --t-srs EPSG:32631 --tr 360 360 {dem4326} {out}.tif
warp_cubic warp -r cubic --t-srs EPSG:32631 --tr 360 360 {dem4326} {out}.tif

# Raster analysis.
polygonize polygonize {classes} {out}.fgb

resample translate --outsize 50% 50% -r bilinear {dem} {out}.tif
